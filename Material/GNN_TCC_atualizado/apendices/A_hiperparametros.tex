\section{Protocolo de avalia\c{c}\~ao estat\'istica}
\label{sec:hp_protocolo}

\begin{table}[H]\centering
\caption{Par\^ametros do protocolo estat\'istico adotado.}
\label{tab:hp_protocolo}
\small
\begin{tabular}{lp{8.5cm}}
\toprule
\textbf{Par\^ametro} & \textbf{Valor / refer\^encia} \\
\midrule
\emph{k}-fold estratificado                & $k = 10$ \cite{kohavi1995cv} \\
Semente do \texttt{StratifiedKFold}        & \texttt{random\_state} $= 42$ \\
Compartilhamento de \emph{folds}           & sim (mesmos \emph{folds} para todos os modelos comparados) \\
Teste de signific\^ancia entre arquiteturas & \texttt{scipy.stats.ttest\_rel} ($t$ pareado por \emph{fold}) \\
Tamanho de efeito                          & Cohen's $d$ \cite{cohen1988statistical, sawilowsky2009effect} \\
\quad Limiar adotado para tese central     & $|d| \geq 0{,}5$ (efeito m\'edio) \\
Bootstrap pareado                           & $n_{\mathrm{boot}} = 2.000$ \cite{efron1993bootstrap} \\
\quad Intervalo de confian\c{c}a            & $95\%$ (percentis $2{,}5$ e $97{,}5$) \\
Varredura de \emph{seeds} (\S\ref{sec:topo_sem_texto}) & 10 \emph{seeds} independentes (42, 123, 456, 789, 1024, 2048, 4096, 8192, 16384, 32768) \\
Crit\'erio de decis\~ao final              & F1-macro (m\'edia sobre \emph{folds}) \\
\bottomrule
\end{tabular}
\end{table}
